class Data extends Component {
  componentDidMount() { this.load(); }              // after first render
  componentDidUpdate(prev) { if (prev.id !== this.props.id) this.load(); }
  componentWillUnmount() { this.abort(); }          // cleanup
  render() { return null; }
}

What to notice

  • Each has an effect-hook equivalent: mount, dependency change, cleanup.
