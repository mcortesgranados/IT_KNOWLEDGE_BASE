const withTimer = (C) => (props) => {
  const [t, setT] = useState(0);
  useEffect(() => { const i = setInterval(() => setT(x => x + 1), 1000); return () => clearInterval(i); }, []);
  return <C {...props} elapsed={t} />;
};

What to notice

  • Injects behaviour as props; wrapper components and hooks cover most of its uses today.
