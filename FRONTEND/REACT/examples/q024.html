import { lazy, Suspense, memo, useMemo } from 'react';

const HeavyChart = lazy(() => import('./HeavyChart'));   // 1. code splitting

const Row = memo(function Row({ item }) {               // 2. skip unchanged rows
  return <li>{item.label}</li>;
});

export default function List({ items, filter }) {
  const visible = useMemo(                          // 3. cache expensive work
    () => items.filter(i => i.label.includes(filter)),
    [items, filter]
  );

  return (
    <Suspense fallback={<p>Loading chart…</p>}>
      <HeavyChart />
      <ul>{visible.map(i => <Row key={i.id} item={i} />)}</ul>
    </Suspense>
  );
}

What to notice

  • Profile first with React DevTools' Profiler — most 'slow' apps are rendering too much, not rendering slowly.
  • `memo` only helps if props stay referentially equal; otherwise it adds cost.
  • For thousands of rows, virtualise the list instead of memoising every row.
